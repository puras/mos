产品介绍

用 MoS 写出你的第一个 Agent

全程可复制可编译：装依赖 → 装配模型与工具 → 跑一轮 → 加自己的工具。

当前为开发阶段：协议层 moproto / mog 尚未发布到可达仓库， 依赖需要本地 replace（见第 1 节）。协议层发布后即可直接 go get。

00前置条件

需要什么说明
Go 1.26+见仓库 go.mod 的 go 指令
模型凭据OpenAI / Anthropic / Gemini 任一，或自建网关（BaseURL）
模型名账号实际可用的模型名，例如 gpt-4o-mini
三个仓库开发阶段需要：mos（本仓库）、moproto、mog 放同一父目录

01添加依赖

建一个新工程：

# 三个仓库放在同一父目录（开发阶段）
git clone https://github.com/puras/mos.git
# 再 clone 协议层 moproto 与 mog 到同级目录

mkdir myagent && cd myagent
go mod init myagent

写 go.mod（三个 replace 都要写）：

module myagent

go 1.26

require github.com/puras/mos v0.0.0

replace github.com/puras/mos => ../mos
replace github.com/puras/moproto v0.1.0 => ../moproto
replace github.com/puras/mog v0.1.6 => ../mog
为什么必须自己写 replace：Go 只认主模块的 replace，依赖模块里的 replace 不会传递。 少了 moproto / mog 两行，go mod tidy 会去公网找这两个模块并失败。

拉依赖并验证（此时还没有代码，能过就行）：

go mod tidy
go build ./...

协议层发布后，删掉三个 replace，改成一行即可：

go get github.com/puras/mos

02第一个 Agent

新建 main.go，整段复制：

package main

import (
	"context"
	"fmt"
	"os"
	"strings"

	"github.com/puras/mos"
	"github.com/puras/mos/llm"
	"github.com/puras/mos/session"
	"github.com/puras/mos/tool"
	"github.com/puras/mos/tool/builtin"
)

func main() {
	// 协议名：openai / anthropic / gemini
	provider := env("MOS_PROVIDER", "openai")
	keyEnv := mos.APIKeyEnv(provider) // openai→OPENAI_API_KEY，框架不读环境变量，只告诉你名字
	if keyEnv == "" {
		fmt.Fprintf(os.Stderr, "未知协议 %q\n", provider)
		os.Exit(1)
	}

	// ① 模型：协议 + 凭据 + 模型名；重试策略由框架执行
	model, err := mos.NewModel(mos.ModelConfig{
		Provider: provider,
		APIKey:   os.Getenv(keyEnv),
		BaseURL:  env("MOS_BASE_URL", ""), // 走网关时填
		Model:    env("MOS_MODEL", "gpt-4o-mini"),
		Retry:    llm.RetryPolicy{MaxAttempts: 3},
	})
	if err != nil {
		fail(err)
	}

	// ② 工具：默认工具 + 一整套文件/搜索工具（jail 根 = 当前目录）
	reg := tool.NewRegistry()
	for _, t := range builtin.Default() {
		if err := reg.Register(t); err != nil {
			fail(err)
		}
	}
	tools, err := builtin.ToolsIn(".")
	if err != nil {
		fail(err)
	}
	for _, t := range tools {
		if err := reg.Register(t); err != nil {
			fail(err)
		}
	}

	// ③ 装配：一次装好模型、工具、安全、上下文工程
	a, err := mos.New(mos.Config{Model: model, Tools: reg})
	if err != nil {
		fail(err)
	}
	fmt.Print(a.Describe()) // 装配快照：装了什么、没装什么

	// ④ 会话（JSONL 持久化，重启可恢复）+ 跑一轮
	sess, err := session.Open("data", "demo")
	if err != nil {
		fail(err)
	}
	q := strings.Join(os.Args[1:], " ")
	if q == "" {
		q = "只回答：1+1=?"
	}
	reply, err := a.Run(context.Background(), sess, q)
	if err != nil {
		fail(err)
	}
	fmt.Println(reply)
	fmt.Printf("[tokens] 共 %d\n", sess.TotalUsage().TotalTokens)
}

func env(k, def string) string {
	if v := strings.TrimSpace(os.Getenv(k)); v != "" {
		return v
	}
	return def
}

func fail(err error) {
	fmt.Fprintln(os.Stderr, err)
	os.Exit(1)
}

跑起来

export MOS_PROVIDER=openai
export OPENAI_API_KEY=sk-...
export MOS_MODEL=gpt-4o-mini

go run . "只回答：1+1=?"

先打印一段装配快照（模型、工具、上下文预算、降级警告），再打印模型回答。

03写自己的工具

工具就是一个实现了 tool.Tool 的结构体：名字、说明、参数类型、执行函数。

package main

import (
	"context"
	"encoding/json"

	"github.com/puras/mos/tool"
)

// 参数：struct tag → JSON Schema（注册时生成并缓存）
type WeatherArgs struct {
	City string `json:"city" mos:"desc=城市名,required=true"`
	Unit string `json:"unit,omitempty" mos:"desc=温度单位：celsius 或 fahrenheit"`
}

type Weather struct {}

func (Weather) Name() string        { return "weather" }
func (Weather) Description() string { return "查询指定城市的当前天气" }
func (Weather) Parameters() any     { return &WeatherArgs{} } // 反射生成 schema 用

func (Weather) Execute(ctx context.Context, raw json.RawMessage) (any, error) {
	args, err := tool.Parse[WeatherArgs](raw)
	if err != nil {
		return nil, err
	}
	// 返回值会被序列化后回填给模型 —— 保持 JSON 可序列化
	return map[string]any{
		"city":   args.City,
		"temp_c": 26,
		"sky":    "晴",
	}, nil
}

在装配处注册（Weather{} 满足 tool.Tool）：

reg := tool.NewRegistry()
if err := reg.Register(Weather{}); err != nil {
	fail(err)
}
  • 工具名就是模型调用时的函数名，重复注册会报错（fail-closed，不静默覆盖）
  • 参数用 struct tag 描述：mos:"desc=说明,required=true"，json tag 决定字段名
  • 返回 error 不会中断整轮——错误会作为工具结果回填给模型，让它自己纠正
  • 默认独占执行。想与同批调用并发，实现 ConcurrencySafe(args json.RawMessage) bool 并返回 true，否则一律串行

04会话与持久化

session.Open(dir, id) 一行建/恢复：日志是 append-only 的 JSONL，落在 dir/id/。

// 多轮对话：同一个 sess 复用，上下文自动接续
sess, _ := session.Open("data", "demo")
for {
	q := readLine("> ")
	if q == "" {
		break
	}
	reply, err := a.Run(context.Background(), sess, q)
	if err != nil {
		fail(err)
	}
	fmt.Println(reply)
}
  • 进程重启、Open 同一个 id → 上下文接着上次继续（崩溃恢复只补事件、不改写历史）
  • sess.Events() 拿到全链路事件（turn / assistant / tool / approval / compaction）
  • sess.TotalUsage() 拿跨轮累计用量（输入、缓存读、输出）
  • 不想落盘：session.New(id, nil) 用内存会话

05权限与安全

默认就是收紧的：fail-closed——未知工具拒绝、缺能力即报错、审批未配置即拒绝。

① 文件边界：工具只能碰 jail 根以内

// 只能访问 /srv/work 及其子目录，越界路径被拒（symlink 逃逸也挡）
tools, err := builtin.ToolsIn("/srv/work")

想运行期加根（用户在会话中途授权另一个目录）：

roots := filesystem.NewRootSet([]string{"/srv/work"}, []string{"/srv/work"})
tools, err := builtin.ToolsInRoots(roots)

a, _ := mos.New(mos.Config{Model: model, Tools: reg, Roots: roots})
// 之后随时加根，对后续调用立即生效
a.AddRoot("/srv/data")

② 名字级策略：谁能被调用、谁能被看见

import "github.com/puras/mos/agent/policy"

a, _ := mos.New(mos.Config{
	Model: model,
	Tools: reg,
	Policy: &policy.Config{
		Allow: []string{"read*", "grep*", "weather"}, // 支持 * 通配
		Deny:  []string{"write", "edit"},                 // 拒绝优先于放行
	},
})

Allow 为 nil = 不限制；空集 = 全禁。被拒的工具不会出现在模型看到的清单里。

③ 人工审批：某些调用要人点头

import "github.com/puras/mos/agent/approval"

type termApprover struct {}

func (termApprover) Decide(ctx context.Context, req approval.Request) (approval.Outcome, error) {
	fmt.Printf("需要确认：%s（%s）\n允许吗？[y/N] ", req.ToolName, req.Reason)
	var s string
	fmt.Scanln(&s)
	if s == "y" || s == "Y" {
		return approval.Approved, nil
	}
	return approval.Rejected, nil
}

a, _ := mos.New(mos.Config{
	Model:    model,
	Tools:    reg,
	Approval: approval.NewService(termApprover{}, nil), // 第二个参数是审计落地口
})

审批有四档结果：Approved / Rejected / Cancelled / Unavailable。 没装配应答方 → 需要确认的调用一律拒绝，不会偷偷放行。

④ 参数级规则：按参数决定放行 / 拒绝 / 转人工

a, _ := mos.New(mos.Config{
	Model: model,
	Tools: reg,
	Rules: myRules, // []policy.Check：deny 进守卫，ask 转审批
})

例如「写文件到某个路径 → 转人工确认」。写法见 agent/policy 包与 examples/chat/guard.go。

06常见问题

go mod tidy 报找不到 moproto / mog

replace 没写在你的 go.mod 里。依赖模块的 replace 不传递，三个都要自己写。

401 / model not found

检查三件事：协议名（MOS_PROVIDER）与密钥变量名是否匹配（mos.APIKeyEnv 返回的名字）、 模型名是否是账号可用的、是否走了网关（MOS_BASE_URL）。

写文件报「未读过该文件」

写前观察门：没读过的文件不允许覆盖，先 read 再 write。这是防丢失更新，不是 bug。

读文件报「越界」

文件工具的 jail 根之外一律拒绝。改根：builtin.ToolsIn("/你的目录")，或运行期 a.AddRoot(dir)。

模型调我的工具被拒绝了

策略（Policy）或守卫拒绝。默认 fail-closed：先看 a.Describe() 打印的装配快照， 再看会话日志里的拒绝事件。

想让模型跑 shell 命令

mos.Config.Sandbox 非 nil 时装配层才注册 shell 工具（sandbox.Local）。 macOS 上会走 Seatbelt 路径强制，需要本机授权；沙箱是纵深防御的一层，不是强隔离边界。

上下文长了变慢、变贵

压缩与截断默认已开（工具结果超限会落盘、超过 token 预算会生成摘要）。 把 ContextWindow 设成模型真实窗口，否则要等溢出才触发压缩。

跑 go test ./... 有失败

沙箱 / shell 类测试需要本机允许 Seatbelt 与本地端口监听的环境，在受限环境下会失败。

07API 速查

函数用途
mos.NewModel(cfg ModelConfig)装配模型：Provider / APIKey / BaseURL / Model / Retry / Log
mos.APIKeyEnv(provider)该协议的密钥环境变量名
mos.New(cfg Config)装配 Agent：Model、Tools、Policy、Rules、Approval、Sandbox、Roots、Context、Prompt…
(a *Agent) Run(ctx, sess, input)跑一轮，返回模型回答
(a *Agent) Describe()装配快照：装了什么、没装什么、降级警告
(a *Agent) AddRoot(dir)运行期加文件根（需装配 Roots）
tool.NewRegistry()建工具注册表
(r *Registry) Register(t)注册工具（进模型的工具清单）
(r *Registry) RegisterDeferred(t)懒加载注册：模型先看不到，经 tool_search 激活
tool.Parse[T](raw)解析模型传来的工具参数
builtin.Default()内置无副作用工具集
builtin.ToolsIn(root)文件 + 搜索工具集，jail 根 = root
builtin.ToolsInRoots(rs)同上，根来自可变集合（支持运行期加根）
session.Open(dir, id)建/恢复 JSONL 会话
(s *Session) Events()全链路事件
(s *Session) TotalUsage()跨轮累计用量
policy.Config{Allow, Deny}名字级策略（支持 * 通配）
approval.NewService(ans, audit)装配审批通道
filesystem.NewRootSet(read, write)可变文件根集合
