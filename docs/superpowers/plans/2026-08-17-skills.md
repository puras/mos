```go
func parseSkillFile(path string) (Skill, bool) {
    b, err := os.ReadFile(path)
    if err != nil {
        return Skill{}, false
    }
    // SKILL.md = `---<yaml>---\n<markdown body>`
    // 用 yaml.Unmarshal 解析首段 frontmatter（遇到非键值对行停止）
    meta := struct {
        Name          string   `yaml:"name"`
        Description   string   `yaml:"description"`
        SchemaVersion string   `yaml:"schema_version"`
        WhenToUse     string   `yaml:"when_to_use"`
        AllowedTools  []string `yaml:"allowed-tools"`
        References    []string `yaml:"references"`
        Paths         []string `yaml:"paths"`
        Source        string   `yaml:"source"`
        UserInvocable bool     `yaml:"user-invocable"`
    }{}
    if err := yaml.Unmarshal(extractFrontmatter(b), &meta); err != nil {
        return Skill{}, false
    }
    s := Skill{
        Name:          strings.TrimSpace(meta.Name),
        Description:   strings.TrimSpace(meta.Description),
        SchemaVersion: strings.TrimSpace(meta.SchemaVersion),
        WhenToUse:     strings.TrimSpace(meta.WhenToUse),
        AllowedTools:  meta.AllowedTools,
        References:    meta.References,
        Paths:         meta.Paths,
        Source:        strings.TrimSpace(meta.Source),
        UserInvocable: meta.UserInvocable,
        Dir:           filepath.Dir(path),
    }
    if s.Name == "" {
        return Skill{}, false
    }
    return s, true
}

// extractFrontmatter 提取 `---` 之间的 YAML 块（含末尾 ---）。
func extractFrontmatter(b []byte) []byte {
    s := string(b)
    if !strings.HasPrefix(strings.TrimSpace(s), "---") {
        return nil
    }
    rest := strings.TrimPrefix(strings.TrimSpace(s), "---")
    end := strings.Index(rest, "\n---")
    if end < 0 {
        return nil
    }
    return []byte(rest[:end])
}
```

import (

// 激活态 ctx 注入：本轮已激活的 Skill 列表 + 工具子集并集。

// ActivatedSkillsFromCtx 返回 ctx 中累积的已激活 Skill 列表；无则返回 nil。
func ActivatedSkillsFromCtx(ctx context.Context) []Skill {
    v, _ := ctx.Value(activatedSkillsKey{}).([]Skill)

// WithActivatedSkills 把 Skill 列表注入 ctx。调用方负责累积（不复制）。
func WithActivatedSkills(ctx context.Context, sk []Skill) context.Context {
    return context.WithValue(ctx, activatedSkillsKey{}, sk)

// ToolSubsetFromCtx 返回 ctx 中累积的允许工具名并集；nil = 未限制。
func ToolSubsetFromCtx(ctx context.Context) []string {
    v, _ := ctx.Value(toolSubsetKey{}).([]string)
    return v
}

// WithToolSubset 把工具名并集注入 ctx；调用方负责并集去重。
func WithToolSubset(ctx context.Context, names []string) context.Context {
    return context.WithValue(ctx, toolSubsetKey{}, names)
}

```go
// renderSkills 把技能清单渲染为两段：
// - 可用技能（命中当前 working dir 的 paths glob）
// - 未命中路径（paths 未匹配，activate_skill 仍可手动调用）
// 无 paths 字段视为全命中（不渲染 fallback 段）。
func renderSkills(skills []Skill) string {
    var matched, unmatched []Skill
    for _, s := range skills {
        if s.PathMatch || len(s.Paths) == 0 {
            matched = append(matched, s)
        } else {
            unmatched = append(unmatched, s)
        }
    }
    var b strings.Builder
    b.WriteString("以下技能按需触发（用 activate_skill 工具激活；激活后正文 + references 路径返回给你）：\n")
    if len(matched) > 0 {
        b.WriteString("\n可用技能：\n")
        for _, s := range matched {
            writeSkillLine(&b, s)
        }
    }
    if len(unmatched) > 0 {
        b.WriteString("\n未命中路径（模型可手动 activate_skill 调用）：\n")
        for _, s := range unmatched {
            writeSkillLine(&b, s)
        }
    }
    return strings.TrimSpace(b.String())
}

func writeSkillLine(b *strings.Builder, s Skill) {
    b.WriteString("- ")
    b.WriteString(s.Name)
    if d := strings.TrimSpace(s.Description); d != "" {
        b.WriteString(": ")
        b.WriteString(d)
    }
    b.WriteString("\n")
}
```

func (a *defaultSystemAssembler) Assemble(ctx context.Context, sess *session.Session, tools []llm.Tool) ([]SystemSection, error) {

```go
package assistant

import (
    "context"
    "encoding/json"
    "fmt"
    "os"
    "path/filepath"
    "strings"

    "hegq.net/mos/agent"
    "hegq.net/mos/agent/event"
    "hegq.net/mos/tool"
)

type ActivateSkill struct {
    Skills []agent.Skill
}

type activateSkillArgs struct {
    Name string `json:"name" jsonschema:"required"`
}

type SkillActivationResult struct {
    Name         string   `json:"name"`
    Body         string   `json:"body"`
    References   []string `json:"references"`
    AllowedTools []string `json:"allowed_tools"`
}

func (t ActivateSkill) Execute(ctx context.Context, args json.RawMessage) (any, error) {
    var a activateSkillArgs
    if err := tool.Parse(args, &a); err != nil {
        return nil, fmt.Errorf("activate_skill: %w", err)
    }
    name := strings.TrimSpace(a.Name)
    if name == "" {
        return nil, fmt.Errorf("activate_skill: name required")
    }
    var target *agent.Skill
    for i := range t.Skills {
        if t.Skills[i].Name == name {
            target = &t.Skills[i]
            break
        }
    }
    if target == nil {
        return nil, fmt.Errorf("activate_skill: skill %q not found", name)
    }

    body, _ := readSkillBody(target.Dir)

    // ctx 注入：本轮已激活 + 工具子集并集
    activated := append(agent.ActivatedSkillsFromCtx(ctx), *target)
    ctx = agent.WithActivatedSkills(ctx, activated)
    existing := agent.ToolSubsetFromCtx(ctx)
    subset := unionStrings(existing, target.AllowedTools)
    ctx = agent.WithToolSubset(ctx, subset)

    // 写 skill/activate 事件
    sess := sessionFromCtx(ctx)
    if sess != nil {
        sess.Append(event.SessionEventSkillActivate, event.MarshalData(event.SkillActivateData{
            Name:         target.Name,
            AllowedTools: target.AllowedTools,
            References:   target.References,
        }))
    }

    return SkillActivationResult{
        Name:         target.Name,
        Body:         body,
        References:   target.References,
        AllowedTools: target.AllowedTools,
    }, nil
}

// readSkillBody 读 SKILL.md 文件的正文部分（frontmatter 之外）。
func readSkillBody(dir string) (string, error) {
    b, err := os.ReadFile(filepath.Join(dir, "SKILL.md"))
    if err != nil {
        return "", err
    }
    s := string(b)
    if !strings.HasPrefix(strings.TrimSpace(s), "---") {
        return strings.TrimSpace(s), nil
    }
    rest := strings.TrimPrefix(strings.TrimSpace(s), "---")
    end := strings.Index(rest, "\n---")
    if end < 0 {
        return strings.TrimSpace(s), nil
    }
    return strings.TrimSpace(rest[end+4:]), nil
}

func unionStrings(a, b []string) []string {
    seen := make(map[string]bool)
    var out []string
    for _, s := range a {
        if !seen[s] {
            seen[s] = true
            out = append(out, s)
        }
    }
    for _, s := range b {
        if !seen[s] {
            seen[s] = true
            out = append(out, s)
        }
    }
    return out
}

// sessionFromCtx 从 ctx 取 *session.Session（装配层注入；nil 时跳过事件）
func sessionFromCtx(ctx context.Context) *session.Session {
    v, _ := ctx.Value(sessionKey{}).(*session.Session)
    return v
}

type sessionKey struct{}
```
